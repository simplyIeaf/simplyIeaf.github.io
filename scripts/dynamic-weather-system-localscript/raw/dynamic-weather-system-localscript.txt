local RAIN_DROP_COUNT = 200
local RAIN_HEIGHT = 30
local RAIN_SPREAD = 50
local rainDrops = {}

for i = 1, RAIN_DROP_COUNT do
	local topAtt = Instance.new("Attachment")
	local bottomAtt = Instance.new("Attachment")
	local startHeight = math.random() * RAIN_HEIGHT
	local x = (math.random() - 0.5) * RAIN_SPREAD
	local z = (math.random() - 0.5) * RAIN_SPREAD
	local length = 1.1 + math.random() * 0.6

	topAtt.Position = Vector3.new(x, startHeight, z)
	bottomAtt.Position = Vector3.new(x, startHeight - length, z)
	topAtt.Parent = dropZone
	bottomAtt.Parent = dropZone

	local beam = Instance.new("Beam")
	beam.Attachment0 = topAtt
	beam.Attachment1 = bottomAtt
	beam.Width0 = 0.04
	beam.Width1 = 0.01
	beam.FaceCamera = true
	beam.LightEmission = 0.25
	beam.Color = ColorSequence.new(Color3.fromRGB(190, 210, 230))
	beam.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.3),
		NumberSequenceKeypoint.new(1, 0.85),
	})
	beam.Enabled = false
	beam.Parent = dropZone

	rainDrops[i] = {
		top = topAtt,
		bottom = bottomAtt,
		beam = beam,
		length = length,
		speed = 55 + math.random() * 25,
	}
end

local rainConnection = nil

local function stepRain(dt)
	for _, drop in ipairs(rainDrops) do
		local pos = drop.top.Position
		local newY = pos.Y - drop.speed * dt
		if newY < -5 then
			newY = RAIN_HEIGHT
		end
		drop.top.Position = Vector3.new(pos.X, newY, pos.Z)
		drop.bottom.Position = Vector3.new(pos.X, newY - drop.length, pos.Z)
	end
end

local function setRainActive(active)
	if active then
		if rainConnection then
			return
		end
		for _, drop in ipairs(rainDrops) do
			drop.beam.Enabled = true
		end
		rainConnection = RunService.Heartbeat:Connect(stepRain)
	else
		if rainConnection then
			rainConnection:Disconnect()
			rainConnection = nil
		end
		for _, drop in ipairs(rainDrops) do
			drop.beam.Enabled = false
		end
	end
end

local snowConnection = nil

local function stepSnow(dt)
	local t = os.clock()
	for _, flake in ipairs(snowFlakes) do
		local pos = flake.top.Position
		local newY = pos.Y - flake.speed * dt
		if newY < -5 then
			newY = SNOW_HEIGHT
		end
		local swayX = flake.baseX + math.sin(t * 0.8 + flake.phase) * flake.sway
		local swayZ = flake.baseZ + math.cos(t * 0.6 + flake.phase) * flake.sway
		flake.top.Position = Vector3.new(swayX, newY, swayZ)
		flake.bottom.Position = Vector3.new(swayX, newY - flake.size, swayZ)
	end
end

local WIND_DIRECTION = Vector3.new(25, 0, 6).Unit
local BASE_WIND_ACCEL = 26
local HEADWIND_ACCEL = 42
local windForce = nil
local windHumanoid = nil

local function setupWindForce(character)
	local root = character:WaitForChild("HumanoidRootPart", 5)
	local humanoid = character:WaitForChild("Humanoid", 5)
	if not root or not humanoid then
		return
	end

	windHumanoid = humanoid

	local attachment = Instance.new("Attachment")
	attachment.Name = "WeatherWindAttachment"
	attachment.Parent = root

	windForce = Instance.new("VectorForce")
	windForce.Name = "WeatherWindForce"
	windForce.Attachment0 = attachment
	windForce.RelativeTo = Enum.ActuatorRelativeTo.World
	windForce.Force = Vector3.new(0, 0, 0)
	windForce.Parent = root
end

RunService.Heartbeat:Connect(function()
	if not (windForce and windForce.Parent) then
		return
	end

	if not windActive then
		windForce.Force = Vector3.new(0, 0, 0)
		return
	end

	local mass = windForce.Parent.AssemblyMass
	local gust = 1 + math.sin(os.clock() * 2.5) * 0.35
	local force = WIND_DIRECTION * mass * BASE_WIND_ACCEL * gust

	if windHumanoid then
		local moveDir = windHumanoid.MoveDirection
		if moveDir.Magnitude > 0.1 then
			local facing = moveDir.Unit:Dot(WIND_DIRECTION)
			if facing < 0 then
				local strength = -facing
				force += (-moveDir.Unit) * mass * HEADWIND_ACCEL * strength
			end
		end
	end

	windForce.Force = force
end)

	rain = "rbxassetid://1516791621",

	wind = "rbxassetid://5799870105",
	thunder = "rbxassetid://1516791621",

local function setAmbientSound(weatherName)
	local id = ambientIds[weatherName]
	if not id then
		TweenService:Create(ambientSound, TweenInfo.new(1.5), {Volume = 0}):Play()
		task.delay(1.6, function()
			if ambientSound.Volume <= 0.02 then
				ambientSound:Stop()
			end
		end)
		return
	end
	if ambientSound.SoundId ~= id then
		ambientSound.SoundId = id
	end
	if not ambientSound.IsPlaying then
		ambientSound:Play()
	end
	TweenService:Create(ambientSound, TweenInfo.new(1.5), {Volume = 0.6}):Play()
end

local thunderConnection = nil

local function playThunderStrike()
	local isNear = math.random() < 0.35
	local peak = isNear and 0.15 or 0.55
	local fadeTime = isNear and 0.6 or 1.1

	flashFrame.BackgroundTransparency = peak
	TweenService:Create(flashFrame, TweenInfo.new(fadeTime, Enum.EasingStyle.Quad), {
		BackgroundTransparency = 1,
	}):Play()

	local rumble = Instance.new("Sound")
	rumble.SoundId = "rbxassetid://6994934244"
	rumble.Volume = isNear and 1 or 0.5
	rumble.PlaybackSpeed = 0.9 + math.random() * 0.3
	rumble.Parent = SoundService
	rumble:Play()
	Debris:AddItem(rumble, 6)
end

local function startThunderLoop()
	local nextStrike = os.clock() + 2 + math.random() * 3
	thunderConnection = RunService.Heartbeat:Connect(function()
		if os.clock() >= nextStrike then
			playThunderStrike()
			nextStrike = os.clock() + 3 + math.random() * 6
		end
	end)
end

local function stopThunderLoop()
	if thunderConnection then
		thunderConnection:Disconnect()
		thunderConnection = nil
	end
	flashFrame.BackgroundTransparency = 1
end

local activeTagsRef = {}

local function applyWeather(weatherName)
	if not presets[weatherName] then
		weatherName = "sun"
	end

	setRainActive(weatherName == "rain" or weatherName == "thunder")
	setSnowActive(weatherName == "snow")

	windActive = weatherName == "wind" or weatherName == "thunder"
	windEmitter.Enabled = weatherName == "wind"
	windEmitter.Rate = weatherName == "wind" and 55 or 0

	if weatherName == "thunder" then
		if not thunderConnection then
			startThunderLoop()
		end
	else
		stopThunderLoop()
	end

	tweenLighting(presets[weatherName], 3)
	setAmbientSound(weatherName)

	for key, tag in pairs(activeTagsRef) do
		tag.Text = key == weatherName and "Active" or ""
	end
end

weatherRemote.OnClientEvent:Connect(applyWeather)

local currentGui = nil

local function destroyWeatherGui()
	if currentGui then
		currentGui:Destroy()
		currentGui = nil
	end
	activeTagsRef = {}
end

local function buildWeatherGui()
	destroyWeatherGui()

	local gui = Instance.new("ScreenGui")
	gui.Name = "WeatherToggleGui"
	gui.ResetOnSpawn = true
	gui.IgnoreGuiInset = true
	gui.Parent = playerGui
	currentGui = gui

	local toggleButton = createIconButton(gui, weatherIconId)
	toggleButton.AnchorPoint = Vector2.new(1, 0.5)
	toggleButton.Position = UDim2.new(1, -30, 0.5, 0)

	local panel, tags = buildWeatherPanel(gui)
	activeTagsRef = tags

	toggleButton.Activated:Connect(function()
		panel.Visible = not panel.Visible
	end)
end

local function onCharacterAdded(character)
	camera = workspace.CurrentCamera
	setupWindForce(character)

	if hasAccess then
		buildWeatherGui()
	end

	local humanoid = character:WaitForChild("Humanoid", 5)
	if humanoid then
		humanoid.Died:Connect(function()
			destroyWeatherGui()
		end)
	end
end

player.CharacterAdded:Connect(onCharacterAdded)

if player.Character then
	onCharacterAdded(player.Character)
end
