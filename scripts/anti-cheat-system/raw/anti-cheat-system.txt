local Plyrs = game:GetService("Players")
local rS = game:GetService("RunService")
local sS = game:GetService("ServerStorage")
local Wrkspc = game:GetService("Workspace")

local Settings = {
	AntiFly = false,
	AntiNoclip = false,
	AntiRootJointDeletion = false,
	AntiMultiTool = false,
	AntiGod = false,
	AntiSpeed = false,
	AntiBuildingTools = false,
	AntiAntiIdle = false,
}

local CHECK_INTERVAL = 0.25
local MAX_HORIZONTAL_SPEED = 40
local MAX_VERTICAL_SPEED = 55
local BASE_LAG_TOLERANCE = 1.15
local PING_TOLERANCE_SCALE = 2
local MAX_LAG_TOLERANCE = 2.25
local ENABLE_KICK = true
local LOG_FLAGS = true

local STRIKE_LIMIT = 3
local RESPAWN_BUFFER = 3
local MAX_AIRTIME = 2.25
local MAX_FLY_HEIGHT = 4
local DEFAULT_HIP_HEIGHT = 3.5

local MAX_EFFECTIVE_HIP = 6
local HARD_MAX_HSPEED = 120
local HARD_MAX_VSPEED = 120
local GRACE_VSPEED_MULT = 1.8
local JUMP_GRACE_COOLDOWN = 1
local WINDOW_SPAN = 3.5
local WINDOW_TOLERANCE = 1.05
local MAX_GRANT_DURATION = 8

local IDLE_SAMPLES = 5
local IDLE_INTERVAL_SPREAD = 0.05
local IDLE_MIN_GAP = 1.5
local IDLE_MOVE_LIMIT = 4

local STRIKE_TOTAL_LIMIT = 8
local STRIKE_REASONS = {
	"SpeedHack",
	"FlyHack",
	"Noclip",
	"RootJointDeletion",
	"MultiTool",
	"BuildingTools",
	"AntiIdle",
	"GodMode",
}

local pS = {}

local rP = RaycastParams.new()
rP.FilterType = Enum.RaycastFilterType.Exclude
rP.RespectCanCollide = true
rP.IgnoreWater = true

local oP = OverlapParams.new()
oP.FilterType = Enum.RaycastFilterType.Exclude

local trrn = Wrkspc.Terrain

local function log(plyr, rsn, extr)
	if not LOG_FLAGS then
		return
	end

	local msg = "[Anti-Cheat] " .. rsn .. " detected for " .. plyr.Name:lower()
	if extr then
		msg = msg .. " - " .. extr
	end
	warn(msg)
end

local function jttr(vl, sprd)
	return vl * (1 + (math.random() * 2 - 1) * sprd)
end

local sR = Instance.new("Folder")
sR.Name = "AntiCheatStrikes"
sR.Parent = sS

local function rSt(vl)
	local cnt, lst, total = vl.Value:match("^(%d+)|([^|]+)|(%d+)$")
	return tonumber(cnt) or 0, tonumber(lst) or 0, tonumber(total) or 0
end

local function wS(vl, cnt, lst, total)
	vl.Value = string.format("%d|%.3f|%d", cnt, lst, total)
end

local function gSF(plyr)
	local nm = tostring(plyr.UserId)
	local fldr = sR:FindFirstChild(nm)

	if not fldr then
		fldr = Instance.new("Folder")
		fldr.Name = nm
		fldr.Parent = sR
	end

	return fldr
end

local function gSV(plyr, rsn)
	local fldr = gSF(plyr)
	local vl = fldr:FindFirstChild(rsn)

	if not vl then
		vl = Instance.new("StringValue")
		vl.Name = rsn
		vl.Value = "0|0|0"
		vl.Parent = fldr
	end

	return vl
end

local function cS(plyr)
	local fldr = sR:FindFirstChild(tostring(plyr.UserId))
	if fldr then
		fldr:Destroy()
	end
end

local function sSt(plyr)
	cS(plyr)

	for _, rsn in ipairs(STRIKE_REASONS) do
		gSV(plyr, rsn)
	end
end

local function aS(plyr, rsn, extr)
	if not pS[plyr] then
		return false
	end

	local vl = gSV(plyr, rsn)
	local cnt, lst, total = rSt(vl)
	local now = os.clock()

	if now - lst > 30 then
		cnt = 0
	end

	cnt += 1
	total += 1
	log(plyr, rsn, extr)

	if cnt >= STRIKE_LIMIT or total >= STRIKE_TOTAL_LIMIT then
		if ENABLE_KICK then
			wS(vl, cnt, now, total)
			pS[plyr] = nil
			plyr:Kick("[Anti-Cheat] " .. rsn .. " detected")
			return true
		end

		warn("[Anti-Cheat] " .. rsn .. " detected for " .. plyr.Name:lower() .. " - kick skipped (kick disabled)")
		cnt = 0
		total = 0
	end

	wS(vl, cnt, now, total)
	return false
end

local function gF(chrctr, rPa, humanoid)
	rP.FilterDescendantsInstances = { chrctr }
	rP.CollisionGroup = rPa.CollisionGroup

	local hH = humanoid.HipHeight > 0 and humanoid.HipHeight or DEFAULT_HIP_HEIGHT
	hH = math.min(hH, MAX_EFFECTIVE_HIP)
	local lngth = rPa.Size.Y / 2 + hH + 1
	local rslt = Wrkspc:Raycast(rPa.Position, Vector3.new(0, -lngth, 0), rP)

	return rslt and rslt.Instance
end

local function nT(chrctr, rPa)
	oP.FilterDescendantsInstances = { chrctr }

	for _, prt in ipairs(Wrkspc:GetPartBoundsInRadius(rPa.Position, 3, oP)) do
		if prt:IsA("TrussPart") then
			return true
		end
	end

	return false
end

local function iW(rPa)
	local pos = rPa.Position
	local rgn = Region3.new(pos - Vector3.new(2, 3, 2), pos + Vector3.new(2, 1, 2)):ExpandToGrid(4)
	local mtrls, occupancies = trrn:ReadVoxels(rgn, 4)
	local sz = mtrls.Size

	for x = 1, sz.X do
		for y = 1, sz.Y do
			for z = 1, sz.Z do
				if mtrls[x][y][z] == Enum.Material.Water and occupancies[x][y][z] > 0 then
					return true
				end
			end
		end
	end

	return false
end

local function iS(chrctr, rPa, humanoid)
	return gF(chrctr, rPa, humanoid) ~= nil or nT(chrctr, rPa) or iW(rPa)
end

local function pO(chrctr, rPa, fP, tP, mdls)
	local drctn = tP - fP
	local dstnc = drctn.Magnitude
	if dstnc < 0.1 then
		return false
	end

	local ignr = { chrctr }
	for _, mdl in ipairs(mdls) do
		if mdl ~= chrctr then
			table.insert(ignr, mdl)
		end
	end

	rP.FilterDescendantsInstances = ignr
	rP.CollisionGroup = rPa.CollisionGroup

	local rslt = Wrkspc:Raycast(fP, drctn, rP)
	if rslt and (rslt.Position - fP).Magnitude < dstnc - 0.5 then
		return true, rslt.Instance.Name
	end

	return false
end

local function gSL(plyr, humanoid)
	local tlrnc = BASE_LAG_TOLERANCE

	local ok, ping = pcall(function()
		return plyr:GetNetworkPing()
	end)

	if ok and type(ping) == "number" then
		tlrnc = math.min(tlrnc + ping * PING_TOLERANCE_SCALE, MAX_LAG_TOLERANCE)
	end

	local cW = math.min(math.max(MAX_HORIZONTAL_SPEED, humanoid.WalkSpeed), HARD_MAX_HSPEED)
	return math.min(cW * tlrnc, HARD_MAX_HSPEED)
end

local function uEF(chrctr, rPa)
	if chrctr:GetAttribute("Ragdolled") then
		return true
	end

	for _, chld in ipairs(rPa:GetChildren()) do
		if
			chld:IsA("BodyMover")
			or chld:IsA("LinearVelocity")
			or chld:IsA("VectorForce")
			or chld:IsA("AlignPosition")
			or chld:IsA("AlignOrientation")
		then
			return true
		end
	end

	return false
end

local function iSe(humanoid)
	return humanoid.SeatPart ~= nil
end

local function gP(plyr)
	local chrctr = plyr.Character
	if not chrctr then
		return nil
	end

	local rPa = chrctr:FindFirstChild("HumanoidRootPart")
	local humanoid = chrctr:FindFirstChildOfClass("Humanoid")
	if not rPa or not humanoid or humanoid.Health <= 0 then
		return nil
	end

	return chrctr, rPa, humanoid
end

local function nS()
	return {
		lastPosition = nil,
		lastCheck = os.clock(),
		lastGroundPos = nil,
		airborneSince = nil,
		floorPart = nil,
		floorPos = nil,
		multiToolChecks = 0,
		flightUntil = 0,
		speedUntil = 0,
		teleportUntil = 0,
		jumpUntil = 0,
		jumpSamples = {},
		wasSeated = false,
		lastJumpGrace = 0,
		windowStart = os.clock(),
		windowDist = 0,
	}
end

local function rT(stt, pstn, flr)
	stt.lastPosition = pstn
	stt.lastCheck = os.clock()
	stt.lastGroundPos = pstn
	stt.airborneSince = nil
	stt.floorPart = flr
	stt.floorPos = flr and flr.Position
	stt.windowStart = os.clock()
	stt.windowDist = 0
end

local function rJ(plyr, stt, rPa)
	local smpls = stt.jumpSamples
	table.insert(smpls, { time = os.clock(), pos = rPa.Position })

	while #smpls > IDLE_SAMPLES do
		table.remove(smpls, 1)
	end

	if not Settings.AntiAntiIdle or #smpls < IDLE_SAMPLES then
		return
	end

	local frst = smpls[1].pos
	local lst = smpls[#smpls].pos
	local mvd = (Vector3.new(lst.X, 0, lst.Z) - Vector3.new(frst.X, 0, frst.Z)).Magnitude
	if mvd > IDLE_MOVE_LIMIT then
		return
	end

	local mG, maxGap = math.huge, 0
	for i = 2, #smpls do
		local gap = smpls[i].time - smpls[i - 1].time
		mG = math.min(mG, gap)
		maxGap = math.max(maxGap, gap)
	end

	if mG >= IDLE_MIN_GAP and maxGap - mG <= IDLE_INTERVAL_SPREAD then
		stt.jumpSamples = {}
		aS(plyr, "AntiIdle", string.format("stationary, jumping every ~%.2fs", (mG + maxGap) / 2))
	end
end

local function tJ(plyr, chrctr, humanoid)
	humanoid.StateChanged:Connect(function(_, nHS)
		if nHS ~= Enum.HumanoidStateType.Jumping then
			return
		end

		local stt = pS[plyr]
		local rPa = chrctr:FindFirstChild("HumanoidRootPart")
		if not stt or not rPa then
			return
		end

		local now = os.clock()
		if iS(chrctr, rPa, humanoid) and now - stt.lastJumpGrace >= JUMP_GRACE_COOLDOWN then
			stt.jumpUntil = now + 0.75
			stt.lastJumpGrace = now
		end

		rJ(plyr, stt, rPa)
	end)
end

local function tD(plyr, chrctr, humanoid)
	humanoid.Died:Connect(function()
		if not Settings.AntiGod or not Plyrs.CharacterAutoLoads then
			return
		end

		task.delay(Plyrs.RespawnTime + RESPAWN_BUFFER, function()
			if plyr.Parent and plyr.Character == chrctr then
				if aS(plyr, "GodMode", "did not respawn after death, forcing respawn") then
					return
				end
				plyr:LoadCharacter()
			end
		end)
	end)
end

local function cRJ(plyr, chrctr, humanoid)
	local rPa = chrctr:FindFirstChild("HumanoidRootPart")
	if not rPa then
		return
	end

	local jnt
	if humanoid.RigType == Enum.HumanoidRigType.R15 then
		local lT = chrctr:FindFirstChild("LowerTorso")
		jnt = lT and lT:FindFirstChild("Root")
	else
		jnt = rPa:FindFirstChild("RootJoint")
	end

	local vld = jnt and jnt:IsA("Motor6D") and jnt.Part0 and jnt.Part1
	if not vld then
		if aS(plyr, "RootJointDeletion", "RootJoint missing or disconnected") then
			return
		end
		humanoid.Health = 0
	end
end

local function cMT(plyr, chrctr, humanoid)
	local stt = pS[plyr]
	if not stt then
		return
	end

	local eqppd = 0
	for _, chld in ipairs(chrctr:GetChildren()) do
		if chld:IsA("Tool") then
			eqppd += 1
		end
	end

	if eqppd <= 1 then
		stt.multiToolChecks = 0
		return
	end

	stt.multiToolChecks += 1
	if stt.multiToolChecks < 2 then
		return
	end

	stt.multiToolChecks = 0
	if aS(plyr, "MultiTool", eqppd .. " tools equipped at once") then
		return
	end
	humanoid:UnequipTools()
end

local function fHB(cntnr)
	if not cntnr then
		return nil
	end

	for _, chld in ipairs(cntnr:GetChildren()) do
		if chld:IsA("HopperBin") then
			return chld
		end
	end

	return nil
end

local function cBT(plyr, chrctr)
	local bckpck = plyr:FindFirstChildOfClass("Backpack")
	local bin = fHB(bckpck) or fHB(chrctr)
	if not bin then
		return
	end

	local nm = bin.Name
	bin:Destroy()
	aS(plyr, "BuildingTools", "HopperBin '" .. nm .. "' present")
end

local function rEC(plyr, chrctr, humanoid)
	if Settings.AntiRootJointDeletion then
		cRJ(plyr, chrctr, humanoid)
	end

	if humanoid.Health > 0 and Settings.AntiMultiTool then
		cMT(plyr, chrctr, humanoid)
	end

	if humanoid.Health > 0 and Settings.AntiBuildingTools then
		cBT(plyr, chrctr)
	end
end

local function uGT(plyr)
	if not Settings.AntiFly then
		return
	end

	local stt = pS[plyr]
	if not stt then
		return
	end

	local chrctr, rPa, humanoid = gP(plyr)
	if not chrctr or iSe(humanoid) then
		return
	end

	if uEF(chrctr, rPa) then
		return
	end

	local now = os.clock()
	if iS(chrctr, rPa, humanoid) then
		stt.lastGroundPos = rPa.Position
		stt.airborneSince = nil
	elseif not stt.airborneSince and now >= stt.flightUntil and now >= stt.jumpUntil then
		stt.airborneSince = now
	end
end

local function cP(plyr, hM)
	local stt = pS[plyr]
	if not stt then
		return
	end

	local chrctr, rPa, humanoid = gP(plyr)
	if not chrctr then
		return
	end

	rEC(plyr, chrctr, humanoid)
	if humanoid.Health <= 0 then
		return
	end

	local now = os.clock()
	local pstn = rPa.Position
	local flr = gF(chrctr, rPa, humanoid)

	if iSe(humanoid) then
		stt.wasSeated = true
		rT(stt, pstn, flr)
		return
	end

	if stt.wasSeated then
		stt.wasSeated = false
		stt.flightUntil = now + 1.25
		stt.speedUntil = now + 1.25
		rT(stt, pstn, flr)
		return
	end

	if now < stt.teleportUntil or uEF(chrctr, rPa) or not stt.lastPosition then
		rT(stt, pstn, flr)
		return
	end

	local dt = now - stt.lastCheck
	if dt <= 0 then
		return
	end

	local dlt = pstn - stt.lastPosition
	if flr and flr == stt.floorPart and flr.Anchored then
		dlt = dlt - (flr.Position - stt.floorPos)
	end

	local hS = Vector3.new(dlt.X, 0, dlt.Z).Magnitude / dt
	local vS = dlt.Y / dt
	local sL = jttr(gSL(plyr, humanoid), 0.08)

	if Settings.AntiSpeed and now >= stt.speedUntil and hS > sL then
		local dtl = string.format("%.1f studs/s (limit %.1f)", hS, sL)
		if aS(plyr, "SpeedHack", dtl) then
			return
		end
	end

	if now >= stt.speedUntil then
		stt.windowDist = (stt.windowDist or 0) + Vector3.new(dlt.X, 0, dlt.Z).Magnitude
	else
		stt.windowStart = now
		stt.windowDist = 0
	end

	local wSpn = now - (stt.windowStart or now)
	if wSpn >= WINDOW_SPAN then
		local avgS = stt.windowDist / math.max(wSpn, 0.01)
		if Settings.AntiSpeed and now >= stt.speedUntil and avgS > sL * WINDOW_TOLERANCE then
			local dtl = string.format("avg %.1f studs/s over %.1fs", avgS, wSpn)
			if aS(plyr, "SpeedHack", dtl) then
				return
			end
		end
		stt.windowStart = now
		stt.windowDist = 0
	end

	local aG = now < stt.flightUntil or now < stt.jumpUntil
	local vC = math.min(aG and MAX_VERTICAL_SPEED * GRACE_VSPEED_MULT or MAX_VERTICAL_SPEED, HARD_MAX_VSPEED)

	if Settings.AntiFly and vS > vC then
		local dtl = string.format("vertical speed %.1f studs/s", vS)
		if aS(plyr, "FlyHack", dtl) then
			return
		end
	end

	if Settings.AntiFly and not aG then
		if stt.airborneSince and stt.lastGroundPos then
			local artm = now - stt.airborneSince
			local hght = pstn.Y - stt.lastGroundPos.Y

			if artm > jttr(MAX_AIRTIME, 0.15) and hght > jttr(MAX_FLY_HEIGHT, 0.2) then
				local dtl = string.format("airborne %.1fs at +%.1f studs", artm, hght)
				if aS(plyr, "FlyHack", dtl) then
					return
				end
				stt.airborneSince = now
			end
		end
	end

	local stdy = dt <= CHECK_INTERVAL * 2 and hS <= sL
	if Settings.AntiNoclip and stdy then
		local blckd, hitName = pO(chrctr, rPa, stt.lastPosition, pstn, hM)
		if blckd then
			if aS(plyr, "Noclip", "passed through '" .. tostring(hitName) .. "'") then
				return
			end
			humanoid.Health = 0
			return
		end
	end

	stt.lastPosition = pstn
	stt.lastCheck = now
	stt.floorPart = flr
	stt.floorPos = flr and flr.Position
end

local AntiCheatAPI = {
	Settings = Settings,
}

local function cGD(drtn)
	return math.clamp(tonumber(drtn) or 0, 0, MAX_GRANT_DURATION)
end

function AntiCheatAPI.GrantTemporaryFlight(plyr, drtn)
	local stt = pS[plyr]
	if stt then
		stt.flightUntil = os.clock() + cGD(drtn)
	end
end

function AntiCheatAPI.GrantTemporarySpeedBoost(plyr, drtn)
	local stt = pS[plyr]
	if stt then
		stt.speedUntil = os.clock() + cGD(drtn)
		stt.windowStart = stt.speedUntil
		stt.windowDist = 0
	end
end

function AntiCheatAPI.GrantTemporaryTeleport(plyr, drtn)
	local stt = pS[plyr]
	if not stt then
		return
	end

	stt.teleportUntil = os.clock() + cGD(drtn)

	local chrctr = plyr.Character
	local rPa = chrctr and chrctr:FindFirstChild("HumanoidRootPart")
	if rPa then
		rT(stt, rPa.Position, nil)
	end
end

_G.AntiCheatAPI = AntiCheatAPI

local function sC(plyr, chrctr)
	if not chrctr then
		return
	end

	local stt = pS[plyr]
	local rPa = chrctr:WaitForChild("HumanoidRootPart", 5)
	if stt and rPa then
		rT(stt, rPa.Position, nil)
	end

	local humanoid = chrctr:WaitForChild("Humanoid", 5)
	if humanoid then
		tJ(plyr, chrctr, humanoid)
		tD(plyr, chrctr, humanoid)
	end
end

local function sP(plyr)
	if pS[plyr] then
		return
	end

	pS[plyr] = nS()
	sSt(plyr)

	plyr.CharacterAdded:Connect(function(chrctr)
		sC(plyr, chrctr)
	end)

	local exstng = plyr.Character
	if exstng then
		task.spawn(sC, plyr, exstng)
	end
end

Plyrs.PlayerAdded:Connect(sP)

Plyrs.PlayerRemoving:Connect(function(plyr)
	pS[plyr] = nil
	cS(plyr)
end)

for _, plyr in ipairs(Plyrs:GetPlayers()) do
	sP(plyr)
end

local elpsd = 0
local nC = CHECK_INTERVAL

rS.Heartbeat:Connect(function(dt)
	local plyrs = Plyrs:GetPlayers()

	for _, plyr in ipairs(plyrs) do
		pcall(uGT, plyr)
	end

	elpsd += dt
	if elpsd < nC then
		return
	end
	elpsd = 0
	nC = jttr(CHECK_INTERVAL, 0.3)

	local hM = {}
	for _, chld in ipairs(Wrkspc:GetChildren()) do
		if chld:FindFirstChildOfClass("Humanoid") then
			table.insert(hM, chld)
		end
	end

	for _, plyr in ipairs(plyrs) do
		local ok, err = pcall(cP, plyr, hM)
		if not ok then
			warn("[Anti-Cheat] Error checking " .. plyr.Name:lower() .. " - " .. tostring(err))
		end
	end
end)
