local function cancelActiveTween()
    if activeFadeTween then
        activeFadeTween:Cancel()
        activeFadeTween = nil
    end
end

local function fadeOut(sound, callback)
    cancelActiveTween()
    local tween = TweenService:Create(sound, fadeOutInfo, {Volume = 0})
    activeFadeTween = tween
    tween:Play()
    
    local conn
    conn = tween.Completed:Connect(function(playbackState)
        if conn then
            conn:Disconnect()
            conn = nil
        end
        if playbackState == Enum.PlaybackState.Completed then
            sound:Stop()
            sound.Volume = 0
            if callback then callback() end
        end
    end)
end

local function fadeIn(sound, isMuted)
    cancelActiveTween()
    sound.Volume = 0
    sound:Play()
    if not isMuted then
        local tween = TweenService:Create(sound, fadeInInfo, {Volume = TARGET_VOLUME})
        activeFadeTween = tween
        tween:Play()
    end
end

local function cleanup()
    for _, conn in ipairs(connections) do
        if conn then conn:Disconnect() end
    end
    connections = {}
    
    if songEndedConnection then
        songEndedConnection:Disconnect()
        songEndedConnection = nil
    end
    
    cancelActiveTween()
    
    if currentSound then
        currentSound:Stop()
        currentSound.Volume = 0
        currentSound = nil
    end
    
    local folder = SoundService:FindFirstChild("MusicFolder")
    if folder then folder:Destroy() end
    
    _G.MusicState = nil
    
    local gui = playerGui:FindFirstChild("MusicInterface")
    if gui then gui:Destroy() end
end

local state = _G.MusicState

local transitionTo
local playRandomSong

local function refreshDisplay()
    for name, tag in pairs(activeTagsRef) do
        tag.Text = (currentSound and currentSound.Name == name) and "Playing" or ""
    end
    if songLabelRef then
        songLabelRef.Text = currentSound and currentSound.Name or "Song Name"
    end
end

transitionTo = function(sound)
    if isTransitioning or sound == nil then return end
    isTransitioning = true
    
    local function begin()
        if songEndedConnection then
            songEndedConnection:Disconnect()
            songEndedConnection = nil
        end
        
        currentSound = sound
        state.currentSoundName = sound.Name
        state.currentTimestamp = 0
        
        fadeIn(sound, state.isMuted)
        refreshDisplay()
        isTransitioning = false
        
        songEndedConnection = sound.Ended:Connect(function()
            playRandomSong()
        end)
    end
    
    if currentSound and currentSound.Playing and currentSound ~= sound then
        fadeOut(currentSound, begin)
    else
        begin()
    end
end

playRandomSong = function()
    local nextSound
    repeat
    nextSound = sounds[math.random(1, #sounds)]
    until nextSound ~= currentSound or #sounds <= 1
    transitionTo(nextSound)
end

local function resumeOrPlay()
    if state.currentSoundName then
        for _, sound in ipairs(sounds) do
            if sound.Name == state.currentSoundName then
                currentSound = sound
                
                if sound.Playing then
                    cancelActiveTween()
                    if not state.isMuted then
                        local tween = TweenService:Create(sound, fadeInInfo, {Volume = TARGET_VOLUME})
                        activeFadeTween = tween
                        tween:Play()
                    end
                else
                    sound.TimePosition = state.currentTimestamp or 0
                    fadeIn(sound, state.isMuted)
                end
                
                if songEndedConnection then
                    songEndedConnection:Disconnect()
                end
                songEndedConnection = sound.Ended:Connect(function()
                    playRandomSong()
                end)
                
                refreshDisplay()
                return
            end
        end
        
        if customSound.Name == state.currentSoundName then
            currentSound = customSound
            
            if customSound.Playing then
                cancelActiveTween()
                if not state.isMuted then
                    local tween = TweenService:Create(customSound, fadeInInfo, {Volume = TARGET_VOLUME})
                    activeFadeTween = tween
                    tween:Play()
                end
            else
                customSound.TimePosition = state.currentTimestamp or 0
                fadeIn(customSound, state.isMuted)
            end
            
            if songEndedConnection then
                songEndedConnection:Disconnect()
            end
            songEndedConnection = customSound.Ended:Connect(function()
                playRandomSong()
            end)
            
            refreshDisplay()
            return
        end
    end
    playRandomSong()
end

local function applyIncoming(payload)
    if type(payload) ~= "table" then return end
    
    if payload.mode == "preset" then
        local target
        for _, s in ipairs(sounds) do
            if s.SoundId == "rbxassetid://" .. tostring(payload.id) then
                target = s
                break
            end
        end
        if target then
            transitionTo(target)
        end
    elseif payload.mode == "custom" then
        customSound.SoundId = "rbxassetid://" .. tostring(payload.id)
        customSound.Name = payload.name or "Custom Track"
        transitionTo(customSound)
    end
end

musicRemote.OnClientEvent:Connect(applyIncoming)

local currentViewportConnection = nil

local function updateScale()
    local cam = workspace.CurrentCamera
    if not cam then return end
    local vp = cam.ViewportSize
    if vp.X <= 0 or vp.Y <= 0 then return end
    local scale = math.min(vp.X / 1920, vp.Y / 1080)
    uiScale.Scale = scale
    safeFrame.Size = UDim2.new(0, vp.X / scale, 0, vp.Y / scale)
end

local function setupCamera()
    if currentViewportConnection then
        currentViewportConnection:Disconnect()
    end
    local cam = workspace.CurrentCamera
    if cam then
        currentViewportConnection = cam:GetPropertyChangedSignal("ViewportSize"):Connect(updateScale)
        updateScale()
    end
end

table.insert(connections, workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(setupCamera))

        lbl.TextXAlignment = Enum.TextXAlignment.Left
        lbl.TextTruncate = Enum.TextTruncate.AtEnd
        lbl.ZIndex = 5
        lbl.Parent = row
        
        local activeTag = Instance.new("TextLabel")
        activeTag.Name = "ActiveTag"
        activeTag.Size = UDim2.new(0.3, 0, 1, 0)
        activeTag.AnchorPoint = Vector2.new(1, 0)
        activeTag.Position = UDim2.new(1, -10, 0, 0)
        activeTag.BackgroundTransparency = 1
        activeTag.Text = ""
        activeTag.TextColor3 = Color3.fromRGB(90, 220, 110)
        activeTag.Font = Enum.Font.FredokaOne
        activeTag.TextSize = 13
        activeTag.TextXAlignment = Enum.TextXAlignment.Right
        activeTag.ZIndex = 5
        activeTag.Parent = row
        
        tags[song.name] = activeTag
        
        local matchedSound
        for _, s in ipairs(sounds) do
            if s.Name == song.name then
                matchedSound = s
                break
            end
        end
        
        if matchedSound then
            row.Activated:Connect(function()
                local now = os.clock()
                if (now - lastPanelAction) < PANEL_COOLDOWN then return end
                lastPanelAction = now
                
                local ok, hasAccess = pcall(function()
                    return musicFunction:InvokeServer()
                end)
                if not ok or not hasAccess then return end
                
                transitionTo(matchedSound)
                musicRemote:FireServer({mode = "preset", id = song.id, name = song.name, scope = currentScope})
            end)
        end
    end
    
    activeTagsRef = tags
    refreshDisplay()
    
    return main
end

local function buildHud(parent)
    local container = Instance.new("Frame")
    container.Name = "MusicHud"
    container.BackgroundTransparency = 1
    container.AnchorPoint = Vector2.new(0, 1)
    container.Position = UDim2.new(0, 40, 1, -40)
    container.Size = UDim2.new(0, 300, 0, 150)
    container.Parent = parent
    
    local listLayout = Instance.new("UIListLayout")
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.VerticalAlignment = Enum.VerticalAlignment.Bottom
    listLayout.HorizontalAlignment = Enum.HorizontalAlignment.Left
    listLayout.Padding = UDim.new(0, 8)
    listLayout.Parent = container
    
    local function createTextElement(className, name, text, layoutOrder)
        local element = Instance.new(className)
        element.Name = name
        element.Text = text
        element.Font = Enum.Font.GothamBold
        element.TextSize = 28
        element.TextColor3 = Color3.new(1, 1, 1)
        element.BackgroundTransparency = 1
        element.TextXAlignment = Enum.TextXAlignment.Left
        element.Size = UDim2.new(1, 0, 0, 35)
        element.LayoutOrder = layoutOrder
        element.Parent = container
        
        local stroke = Instance.new("UIStroke")
        stroke.Color = Color3.new(0, 0, 0)
        stroke.Thickness = 2.5
        stroke.Parent = element
        
        return element
    end
    
    local skipBtn = createTextElement("TextButton", "SkipButton", "Skip", 1)
    local muteBtn = createTextElement("TextButton", "MuteButton", state.isMuted and "Unmute" or "Mute", 2)
    local songLbl = createTextElement("TextLabel", "SongLabel", "Song Name", 3)
    
    table.insert(connections, skipBtn.Activated:Connect(function()
        local now = os.clock()
        if isTransitioning or (now - lastSkipTime) < SKIP_DEBOUNCE then return end
        lastSkipTime = now
        playRandomSong()
    end))
    
    table.insert(connections, muteBtn.Activated:Connect(function()
        state.isMuted = not state.isMuted
        muteBtn.Text = state.isMuted and "Unmute" or "Mute"
        
        if currentSound then
            cancelActiveTween()
            if state.isMuted then
                local tween = TweenService:Create(currentSound, fadeOutInfo, {Volume = 0})
                activeFadeTween = tween
                tween:Play()
            else
                local tween = TweenService:Create(currentSound, fadeInInfo, {Volume = TARGET_VOLUME})
                activeFadeTween = tween
                tween:Play()
            end
        end
    end))
    
    songLabelRef = songLbl
    refreshDisplay()
    
    return container
end

local function destroyToggleAndPanel()
    if currentPanel then
        currentPanel:Destroy()
        currentPanel = nil
    end
    if currentToggleButton then
        currentToggleButton:Destroy()
        currentToggleButton = nil
    end
end

local function buildInterface()
    destroyToggleAndPanel()
    
    local toggleButton = createIconButton(safeFrame, musicIconId)
    toggleButton.AnchorPoint = Vector2.new(1, 1)
    toggleButton.Position = UDim2.new(1, -40, 1, -40)
    currentToggleButton = toggleButton
    
    if not hudBuilt then
        buildHud(safeFrame)
        hudBuilt = true
    end
    
    local panel = buildMusicPanel(safeFrame)
    currentPanel = panel
    
    table.insert(connections, toggleButton.Activated:Connect(function()
        panel.Visible = not panel.Visible
    end))
end

local function onCharacterDied()
    if currentSound then
        state.currentTimestamp = currentSound.TimePosition
    end
    destroyToggleAndPanel()
end

local function onCharacterAdded(character)
    local humanoid = character:WaitForChild("Humanoid")
    humanoid.Died:Connect(onCharacterDied)
    
    if hudBuilt then
        buildInterface()
        resumeOrPlay()
    end
end

player.CharacterAdded:Connect(onCharacterAdded)
if player.Character then
    onCharacterAdded(player.Character)
end

local function requestAccessAndStart()
    local ok, hasAccess, initialState = pcall(function()
        return musicFunction:InvokeServer()
    end)
    
    if ok and hasAccess then
        buildInterface()
    end
    
    if ok and initialState then
        applyIncoming(initialState)
    else
        resumeOrPlay()
    end
end

requestAccessAndStart()
