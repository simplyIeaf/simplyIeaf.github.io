local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")
local MessagingService = game:GetService("MessagingService")

local allowedUserIds = {8033814042, 987654321}

local weatherFolder = Instance.new("Folder")
weatherFolder.Name = "WeatherRemotes"
weatherFolder.Parent = ReplicatedStorage

local weatherRemote = Instance.new("RemoteEvent")
weatherRemote.Name = "WeatherRemote"
weatherRemote.Parent = weatherFolder

local weatherFunction = Instance.new("RemoteFunction")
weatherFunction.Name = "WeatherFunction"
weatherFunction.Parent = weatherFolder

local weatherOrder = {"sun", "rain", "fog", "wind", "snow", "thunder"}
local validWeather = {}
for _, name in ipairs(weatherOrder) do
	validWeather[name] = true
end

local currentWeather = "sun"
local lastRequestAt = {}
local REQUEST_COOLDOWN = 2
local WEATHER_TOPIC = "WeatherSync_v1"
local AUTO_SWAP_INTERVAL = 300
local nextAutoSwapTime = os.clock() + AUTO_SWAP_INTERVAL

local weatherTransitions = {
	sun = {sun = 10, wind = 30, fog = 20, rain = 15, snow = 5, thunder = 3},
	wind = {wind = 15, sun = 20, rain = 25, fog = 15, snow = 10, thunder = 10},
	rain = {rain = 20, thunder = 30, wind = 15, fog = 15, sun = 10, snow = 5},
	fog = {fog = 20, sun = 30, wind = 20, rain = 15, snow = 10, thunder = 5},
	snow = {snow = 25, fog = 20, wind = 15, sun = 15, rain = 10, thunder = 5},
	thunder = {thunder = 15, rain = 35, wind = 20, fog = 10, sun = 10, snow = 5},
}

local function hasAccess(player)
	for _, id in ipairs(allowedUserIds) do
		if player.UserId == id then
			return true
		end
	end
	return false
end

local function broadcastCurrentWeather()
	weatherRemote:FireAllClients(currentWeather)
end

local function applyWeatherLocally(weatherName)
	if not validWeather[weatherName] then
		return
	end
	currentWeather = weatherName
	nextAutoSwapTime = os.clock() + AUTO_SWAP_INTERVAL
	broadcastCurrentWeather()
end

local function publishGlobalWeather(weatherName)
	local success, err = pcall(function()
		MessagingService:PublishAsync(WEATHER_TOPIC, {
			weather = weatherName,
			source = game.JobId,
		})
	end)
	if not success then
		warn("failed to publish weather sync: " .. tostring(err))
		applyWeatherLocally(weatherName)
	end
end

local subscribed, subscribeErr = pcall(function()
	MessagingService:SubscribeAsync(WEATHER_TOPIC, function(message)
		local data = message.Data
		if typeof(data) ~= "table" then
			return
		end
		if typeof(data.weather) ~= "string" then
			return
		end
		if not validWeather[data.weather] then
			return
		end
		applyWeatherLocally(data.weather)
	end)
end)

if not subscribed then
	warn("failed to subscribe to weather topic: " .. tostring(subscribeErr))
end

local function setWeather(weatherName, scope)
	if scope == "global" then
		publishGlobalWeather(weatherName)
	else
		applyWeatherLocally(weatherName)
	end
end

weatherFunction.OnServerInvoke = function(player)
	return hasAccess(player), currentWeather
end

weatherRemote.OnServerEvent:Connect(function(player, payload)
	if not hasAccess(player) then
		return
	end
	if typeof(payload) ~= "table" then
		return
	end

	local weatherName = payload.weather
	local scope = payload.scope
	if typeof(weatherName) ~= "string" or typeof(scope) ~= "string" then
		return
	end

	weatherName = weatherName:lower()
	scope = scope:lower()

	if not validWeather[weatherName] then
		return
	end
	if scope ~= "server" and scope ~= "global" then
		return
	end

	local now = os.clock()
	local last = lastRequestAt[player.UserId]
	if last and now - last < REQUEST_COOLDOWN then
		return
	end
	lastRequestAt[player.UserId] = now

	setWeather(weatherName, scope)
end)

Players.PlayerRemoving:Connect(function(player)
	lastRequestAt[player.UserId] = nil
end)

local function setupPlayer(player)
	weatherRemote:FireClient(player, currentWeather)
end

Players.PlayerAdded:Connect(setupPlayer)

for _, player in ipairs(Players:GetPlayers()) do
	setupPlayer(player)
end

local function pickNaturalWeather()
	local weights = weatherTransitions[currentWeather] or weatherTransitions.sun
	local total = 0
	for _, weight in pairs(weights) do
		total += weight
	end

	local roll = math.random() * total
	local running = 0
	for name, weight in pairs(weights) do
		running += weight
		if roll <= running then
			return name
		end
	end
	return currentWeather
end

task.spawn(function()
	while true do
		task.wait(5)
		if os.clock() >= nextAutoSwapTime then
			local nextWeather = pickNaturalWeather()
			if nextWeather ~= currentWeather then
				applyWeatherLocally(nextWeather)
			else
				nextAutoSwapTime = os.clock() + AUTO_SWAP_INTERVAL
			end
		end
	end
end)
